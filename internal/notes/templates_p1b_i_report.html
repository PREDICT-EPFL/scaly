Function templates P1b-i: sc.function(inputs, output=...)

2026-09-27 · branch claude/function-templates · plan v2, step P1b-i

Summary. The decorator takes its output as output=, the first half of the new signature sc.function(*slots, output=, name=). Still one slot: a body with several parameters arrives in P1b-ii. The old two-positional spelling is refused at decoration with the fix spelled out, so a site the codemod missed cannot run silently. A codemod rewrote all 306 sites; four needed hand edits.

What changed

  • function/api.py: function(inputs, /, *extra, output=None, name=None). At decoration it compares the declared slots with the body's positional parameters. sc.function(IN, OUT) on a one-parameter body raises "sc.function takes one declaration per parameter and the output as output=; did you mean sc.function(<inputs>, output=<outputs>)?". Other mismatches name both counts and the parameters. Parameters with defaults, *args, **kwargs and keyword-only parameters are refused; defaults stay free for static arguments.
  • The codemod inserted output= before the second positional argument of every sc.function/scaly.function call: 306 sites in 85 tracked files (tests 104, examples 88 in .py and 58 in notebooks, benchmarks 29, docs 19, plugin tests 7, README 1). It uses ast positions with UTF-8 byte offsets; notebooks round-trip byte for byte; every edited file and cell re-parses. The script is at the end of this report.
  • Hand edits:
    • the bash heredoc in docs/guide/installation.md;
    • def solve_with_ordering(K, m=m) in sparse_fem_topology.ipynb: the default is dropped. The trace is eager, so the loop variable is read at decoration as before.
    • the two direct sc.Function(name, body, inputs, outputs) constructions in examples/qp_solvers/generated_piqp.py, now the decorator called as a function;
    • the stale sc.Function("f", [x], [y], ...) and f.call in docs/how_it_works/ir.md.

The codemod

"""Dry-run codemod: sc.function(IN, OUT, ...) -> sc.function(IN, output=OUT, ...) over a tree copy.

Usage: uv run codemod.py <root>. Rewrites .py, .ipynb (code cells) and .md (python fences) in place
under <root>; prints per-file counts and anything it refused.
"""

from __future__ import annotations

import ast
import json
import re
import sys
from pathlib import Path

ROOT = Path("/Users/cnjones/git/projects/alloy")
SKIP = ("internal/", "typing_playground/", ".venv/", "third_party/")
MODS = {"sc", "scaly"}


def targets(src: str) -> tuple[list[tuple[int, int]], list[str]]:
  """(line, byte col) insertion points for 'output=' and refusals."""
  tree = ast.parse(src)
  bare = {a.asname or a.name for n in ast.walk(tree) if isinstance(n, ast.ImportFrom) and n.module == "scaly" for a in n.names if a.name == "function"}
  out, refused = [], []
  for node in ast.walk(tree):
    if not isinstance(node, ast.Call):
      continue
    f = node.func
    hit = (isinstance(f, ast.Attribute) and f.attr == "function" and isinstance(f.value, ast.Name) and f.value.id in MODS) or (
      isinstance(f, ast.Name) and f.id in bare
    )
    if not hit:
      continue
    if any(k.arg == "output" for k in node.keywords):
      continue  # already migrated: idempotent
    if len(node.args) != 2 or any(isinstance(a, ast.Starred) for a in node.args):
      refused.append(f"line {node.lineno}: {ast.unparse(node)[:80]}")
      continue
    out.append((node.args[1].lineno, node.args[1].col_offset))
  return out, refused


def apply(lines: list[bytes], edits: list[tuple[int, int]]) -> None:
  for line, col in sorted(edits, reverse=True):
    b = lines[line - 1]
    lines[line - 1] = b[:col] + b"output=" + b[col:]


def do_py(path: Path) -> int:
  src = path.read_bytes()
  edits, refused = targets(src.decode())
  for r in refused:
    print(f"  REFUSED {path}: {r}")
  if edits:
    lines = src.splitlines(keepends=True)
    apply(lines, edits)
    new = b"".join(lines)
    ast.parse(new.decode())
    path.write_bytes(new)
  return len(edits)


def do_ipynb(path: Path) -> int:
  raw = path.read_text(encoding="utf-8")
  nb = json.loads(raw)
  assert json.dumps(nb, indent=1, ensure_ascii=False) + "\n" == raw, f"{path}: JSON would not round-trip"
  n = 0
  for cell in nb["cells"]:
    if cell["cell_type"] != "code":
      continue
    src_lines = cell["source"]
    src = "".join(src_lines)
    masked = "\n".join(("pass" if l.lstrip().startswith(("%", "!")) else l) for l in src.split("\n"))
    edits, refused = targets(masked)
    for r in refused:
      print(f"  REFUSED {path}: {r}")
    if not edits:
      continue
    blines = [l.encode() for l in src.splitlines(keepends=True)]
    apply(blines, edits)
    new = b"".join(blines).decode()
    ast.parse(new)
    cell["source"] = new.splitlines(keepends=True)
    n += len(edits)
  if n:
    path.write_text(json.dumps(nb, indent=1, ensure_ascii=False) + "\n", encoding="utf-8")
  return n


FENCE = re.compile(r"^(\s*)```(python|py)\s*$")


def do_md(path: Path) -> int:
  lines = path.read_bytes().splitlines(keepends=True)
  n = 0
  i = 0
  while i < len(lines):
    m = FENCE.match(lines[i].decode())
    if not m:
      i += 1
      continue
    indent = len(m.group(1).encode())
    j = i + 1
    while j < len(lines) and not re.match(rb"^\s*```\s*$", lines[j]):
      j += 1
    body = b"".join(l[indent:] if l[:indent].isspace() or indent == 0 else l for l in lines[i + 1 : j]).decode()
    try:
      edits, refused = targets(body)
    except SyntaxError as exc:
      if "function(" in body:
        print(f"  UNPARSED {path}:{i + 1}: {exc.msg}")
      i = j + 1
      continue
    for r in refused:
      print(f"  REFUSED {path}:{i + 1}: {r}")
    apply(lines, [(i + 1 + line, indent + col) for line, col in edits])
    n += len(edits)
    i = j + 1
  if n:
    path.write_bytes(b"".join(lines))
  return n


total = {}
import subprocess
TRACKED = subprocess.run(["git", "ls-files"], cwd=ROOT, capture_output=True, text=True).stdout.split("\n")
for path in sorted(ROOT / t for t in TRACKED if t):
  rel = path.relative_to(ROOT).as_posix()
  if not path.is_file() or rel.startswith(SKIP):
    continue
  if path.suffix == ".py":
    k = do_py(path)
  elif path.suffix == ".ipynb":
    k = do_ipynb(path)
  elif path.suffix == ".md":
    k = do_md(path)
  else:
    continue
  if k:
    total[rel] = k
print("files", len(total), "sites", sum(total.values()))
by = {}
for rel, k in total.items():
  top = rel.split("/")[0] if not rel.startswith("plugins") else "/".join(rel.split("/")[:3])
  by[top] = by.get(top, 0) + k
print(by)
